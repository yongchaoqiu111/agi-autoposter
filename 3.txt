#!/usr/bin/env python3
"""
分身调度演示 - multi_scheduler_demo.py

展示核心能力：
  1. 同一个大模型 API 被包装成多个「无状态调度分身」
  2. 所有分身同时向中间层提交任务
  3. 分身之间不通信、不共享状态、互不感知
  4. 中间层统一排队、执行、返回结果

运行前提：
  1. 中间层已启动（python middleware.py）
  2. 网络正常
"""

import asyncio
import httpx
import time
import random
from typing import Dict, List
from datetime import datetime

# 中间层地址
MIDDLEWARE_URL = "http://localhost:8000"

# 分身配置
NUM_SCHEDULERS = 5           # 启动 5 个调度分身
TASKS_PER_SCHEDULER = 3      # 每个分身发布 3 个任务
TASK_INTERVAL_SEC = 0.5      # 分身内任务间隔（秒）


class SchedulerClone:
    """
    调度分身
    - 每个分身是完全独立的实例
    - 不知道其他分身的存在
    - 只负责「发布任务 + 轮询结果」
    - 无状态：所有状态由中间层维护
    """

    def __init__(self, clone_id: int, client: httpx.AsyncClient):
        self.clone_id = clone_id
        self.client = client
        self.tasks_submitted: List[Dict] = []

    async def submit_one_task(self, task_index: int) -> Dict:
        """提交单个任务"""
        platform = random.choice(
            ["douyin", "facebook", "kuaishou", "tiktok", "x", "instagram"]
        )
        txt_id = random.randint(1, 20)
        img_id = random.randint(1, 10)

        task_payload = {
            "platform": platform,
            "txt_id": txt_id,
            "img_id": img_id,
            "max_retries": 2,
        }

        start_time = time.time()
        resp = await self.client.post(f"{MIDDLEWARE_URL}/api/tasks", json=task_payload)
        elapsed = time.time() - start_time

        task_info = {
            "task_id": resp.json()["task_id"],
            "platform": platform,
            "txt_id": txt_id,
            "img_id": img_id,
            "submit_time": datetime.now().isoformat(),
            "submit_duration_ms": round(elapsed * 1000, 2),
        }
        self.tasks_submitted.append(task_info)
        print(f"[分身-{self.clone_id:02d}] 提交任务 {task_info['task_id']} | "
              f"{platform} | txt#{txt_id} img#{img_id} | 耗时 {task_info['submit_duration_ms']}ms")
        return task_info

    async def run(self):
        """分身运行：依次提交所有任务"""
        print(f"\n[分身-{self.clone_id:02d}] 启动，准备提交 {TASKS_PER_SCHEDULER} 个任务")
        for i in range(TASKS_PER_SCHEDULER):
            await self.submit_one_task(i + 1)
            await asyncio.sleep(TASK_INTERVAL_SEC)
        print(f"[分身-{self.clone_id:02d}] 任务提交完毕，提交了 {len(self.tasks_submitted)} 个任务")

    async def poll_results(self):
        """轮询自己的任务结果"""
        results = []
        for task_info in self.tasks_submitted:
            task_id = task_info["task_id"]
            start_time = time.time()
            # 最多轮询 30 秒
            for _ in range(30):
                resp = await self.client.get(f"{MIDDLEWARE_URL}/api/tasks/{task_id}")
                data = resp.json()
                if data["status"] in ["success", "failed", "timeout"]:
                    break
                await asyncio.sleep(1)
            elapsed = time.time() - start_time

            result_info = {
                "task_id": task_id,
                "status": data["status"],
                "result": data.get("result"),
                "poll_duration_sec": round(elapsed, 2),
            }
            results.append(result_info)
            print(f"[分身-{self.clone_id:02d}] 任务 {task_id} 完成 | 状态: {data['status']} | "
                  f"轮询耗时 {elapsed:.2f}s")
        return results


async def main():
    print("=" * 70)
    print("  分身调度演示")
    print("=" * 70)
    print(f"\n配置：")
    print(f"  - 调度分身数量: {NUM_SCHEDULERS}")
    print(f"  - 每个分身任务数: {TASKS_PER_SCHEDULER}")
    print(f"  - 总任务数: {NUM_SCHEDULERS * TASKS_PER_SCHEDULER}")
    print(f"  - 中间层地址: {MIDDLEWARE_URL}")
    print("\n核心验证点：")
    print("  ✅ 多个分身同时运行，互不感知")
    print("  ✅ 中间层统一排队，按顺序执行")
    print("  ✅ 分身无状态，不维护任务队列")
    print()

    async with httpx.AsyncClient(timeout=30.0) as client:
        # 1. 启动所有分身（并发提交任务）
        schedulers = [SchedulerClone(i + 1, client) for i in range(NUM_SCHEDULERS)]
        print("\n[阶段1] 所有分身同时启动，并发提交任务")
        await asyncio.gather(*[s.run() for s in schedulers])

        # 2. 等待中间层完成任务执行
        print("\n[阶段2] 等待中间层执行任务（约 30 秒）...")
        await asyncio.sleep(35)

        # 3. 各分身轮询自己的任务结果
        print("\n[阶段3] 分身轮询任务结果")
        all_results = await asyncio.gather(*[s.poll_results() for s in schedulers])

        # 4. 汇总统计
        total_tasks = 0
        success_count = 0
        for results in all_results:
            total_tasks += len(results)
            success_count += sum(1 for r in results if r["status"] == "success")

        print("\n" + "=" * 70)
        print("  执行汇总")
        print("=" * 70)
        print(f"  总任务数: {total_tasks}")
        print(f"  成功数: {success_count}")
        print(f"  失败/超时: {total_tasks - success_count}")
        print(f"  成功率: {success_count / total_tasks * 100:.1f}%")
        print("\n✅ 分身调度演示完成")
        print("   - 所有分身独立运行，互不感知")
        print("   - 任务由中间层统一排队和调度")
        print("   - 分身无状态，可线性扩展")


if __name__ == "__main__":
    asyncio.run(main())
